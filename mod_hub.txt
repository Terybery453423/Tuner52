// @name         Tuner52 (RALLY52)

(function(){
  'use strict';
  if (window.__r52hub) return; window.__r52hub = 1;

  const HUB_VERSION = '2.0.12'; 
  const REPO = 'https://raw.githubusercontent.com/Terybery453423/Tuner52/main/mods/';
  const ROOT_URL = 'https://raw.githubusercontent.com/Terybery453423/Tuner52/main/mod_hub.txt';
  const ALLOWED = [REPO, ROOT_URL];
  
  let SECRET = localStorage.getItem('r52_secret') === 'true';
  let UNLOCKED = localStorage.getItem('r52_unlocked') === 'true';
  let installed = JSON.parse(localStorage.getItem('r52_installed') || '[]');
  let active = JSON.parse(localStorage.getItem('r52_active') || '{}');
  let modCache = JSON.parse(localStorage.getItem('r52_mod_cache') || '{}'); // BUG 5 FIX
  let mods = [];
  let tab = 'mods';
  let updateAvailable = false;
  let latestVersion = HUB_VERSION;
  let latestScriptCode = '';

  const xhr = unsafeWindow.GM_xmlhttpRequest || GM_xmlhttpRequest;
  function fetchText(url) {
    return new Promise((res, rej) => {
      if (!ALLOWED.some(p => url.startsWith(p))) return rej('URL blocked');
      xhr({ method: 'GET', url: url + '?t=' + Date.now(), timeout: 10000,
        onload: r => r.status >= 200 && r.status < 300 ? res(r.responseText) : rej('HTTP ' + r.status),
        onerror: () => rej('Net error')
      });
    });
  }

  async function checkVersion() {
    try {
      const code = await fetchText(ROOT_URL);
      latestScriptCode = code;
      const match = code.match(/@version\s+([0-9.]+)/);
      if (match && match[1]) {
        latestVersion = match[1];
        if (latestVersion !== HUB_VERSION) {
          updateAvailable = true;
          const mUI = document.getElementById('r52_mods_ui');
          if (mUI && mUI.style.display !== 'none') { mUI.dataset.loaded = '0'; renderMods(mUI); }
        }
      }
    } catch(e) {}
  }

  // BUG 4 FIX: Add a timeout so it doesn't hang silently
  function inject(code) {
    return new Promise((resolve) => {
      let resolved = false;
      const evt = 'r52_inject_' + Math.random().toString(36);
      const s = document.createElement('script');
      s.textContent = code + `\n;document.dispatchEvent(new CustomEvent('${evt}'));`;
      
      document.addEventListener(evt, () => { resolved = true; resolve(); }, { once: true });
      (document.head || document.documentElement).appendChild(s);
      s.remove();

      setTimeout(() => { if (!resolved) resolve(); }, 2000); // Resolve anyway after 2s
    });
  }

  inject(`
    window.__r52DownloadFile = function(url, name, onSuccess, onError) {
      const id = Math.random().toString(36);
      window._r52_dl_resolvers = window._r52_dl_resolvers || {};
      window._r52_dl_resolvers[id] = { onSuccess, onError };
      window.dispatchEvent(new CustomEvent('r52_hub_download', { detail: { url, name, id } }));
    };
    window.__r52ModUIToggles = window.__r52ModUIToggles || {};
    window.__r52ApplyUIToggle = function(id, visible) {
      if (window.__r52ModUIToggles[id]) window.__r52ModUIToggles[id](visible);
    };
  `);

  window.addEventListener('r52_hub_download', (e) => {
    const { url, name, id } = e.detail;
    xhr({ method: 'GET', url: url, responseType: 'blob', timeout: 30000,
      onload: r => {
        if (r.status >= 200 && r.status < 300) {
          const objectUrl = URL.createObjectURL(r.response);
          if (unsafeWindow._r52_dl_resolvers[id]) {
            unsafeWindow._r52_dl_resolvers[id].onSuccess(objectUrl);
            delete unsafeWindow._r52_dl_resolvers[id];
          }
        } else { if (unsafeWindow._r52_dl_resolvers[id]) { unsafeWindow._r52_dl_resolvers[id].onError('HTTP ' + r.status); delete unsafeWindow._r52_dl_resolvers[id]; } }
      },
      onerror: () => { if (unsafeWindow._r52_dl_resolvers[id]) { unsafeWindow._r52_dl_resolvers[id].onError('Net error'); delete unsafeWindow._r52_dl_resolvers[id]; } }
    });
  });

  async function loadMod(m) {
    if (active[m.id] === false) return;
    try {
      const code = await fetchText(m.url);
      await inject(code);
      const isHidden = localStorage.getItem('r52_ui_' + m.id) === 'true';
      if (isHidden) await inject(`if(window.__r52ApplyUIToggle) window.__r52ApplyUIToggle('${m.id}', false);`);
      const aTab = Array.from(document.querySelectorAll('.mi')).find(b => b.textContent === '⚡ ACTIVE');
      if (aTab && aTab.classList.contains('sel')) {
        const mUI = document.getElementById('r52_mods_ui');
        if (mUI) { const list = mUI.querySelector('div[style*="overflow-y:auto"]'); if (list) renderActive(list); }
      }
    } catch(e) { alert('❌ Failed to load: ' + m.name + '\nError: ' + e.message); }
  }

  checkVersion();
  (async () => {
    installed = [...new Set(installed)];
    if (!installed.length) return;
    
    // BUG 5 FIX: Load from cache first, then try to update from store
    for (const id of installed) {
        if (active[id] !== false && modCache[id]) {
            loadMod(modCache[id]);
        }
    }

    try {
      const store = JSON.parse(await fetchText(REPO + 'store.json'));
      mods = mods.concat(store);
      store.forEach(m => { 
          modCache[m.id] = m; // Update cache
          if (installed.includes(m.id) && active[m.id] !== false && !modCache[m.id]._loaded) loadMod(m); 
      });
      if (SECRET) { const sec = JSON.parse(await fetchText(REPO + '45yllar.json')); mods = mods.concat(sec); sec.forEach(m => { modCache[m.id] = m; if (installed.includes(m.id) && active[m.id] !== false) loadMod(m); }); }
      localStorage.setItem('r52_mod_cache', JSON.stringify(modCache));
    } catch(e) {}
  })();

  // BUG 1 FIX: Save original handler and don't clobber
  let taps = 0, tapT, socialsHijacked = false;
  function hijackSocials() {
    if (socialsHijacked) return;
    const tSocial = document.querySelector('.tSocial');
    if (!tSocial) return;
    const MY_TIKTOK = 'https://www.tiktok.com/@tuner52_';
    const MY_DISCORD = 'https://discord.gg/EMFymvnCuC';
    const ig = tSocial.querySelector('a[href*="instagram.com"]'); if (ig) ig.remove();
    const fb = tSocial.querySelector('a[href*="facebook.com"]'); if (fb) fb.remove();
    const tt = tSocial.querySelector('a[href*="tiktok.com"]'); if (tt) { tt.href = MY_TIKTOK; tt.title = "TikTok"; }
    const dc = tSocial.querySelector('a[href*="discord.gg"]') || tSocial.querySelector('a[href*="discord.com"]');
    if (dc) { dc.href = MY_DISCORD; dc.title = "Discord"; }
    socialsHijacked = true;
  }

  let tapInterval = setInterval(() => {
    hijackSocials(); 
    const optsH = document.getElementById('optsH'), optCats = document.getElementById('optCats'), optRows = document.getElementById('optRows');
    if (!optsH || !optCats || !optRows) { tab = 'mods'; return; }
    
    const optsEl = document.getElementById('opts');
    if (optsEl && !optsEl.dataset.zset) { optsEl.style.zIndex = '2147483647'; optsEl.dataset.zset = '1'; }
    
    if (!optsH.dataset.hooked) {
      optsH.dataset.hooked = '1';
      // Save original onclick if it exists
      const origClick = optsH.onclick; 
      optsH.addEventListener('click', (e) => {
        // Call original first
        if (origClick) origClick.call(optsH, e);
        
        taps++; clearTimeout(tapT); tapT = setTimeout(() => taps = 0, 1500); 
        if (taps >= 5) { 
            taps = 0; UNLOCKED = true; localStorage.setItem('r52_unlocked', 'true'); buildTabs(); 
            clearInterval(tapInterval); // BUG 6 FIX: Stop interval once unlocked
        }
      });
    }
    if (!optCats.dataset.hooked) {
      optCats.dataset.hooked = '1';
      optCats.addEventListener('click', e => { if (!e.target.closest('#r52_mods_btn') && !e.target.closest('#r52_fixes_btn')) tab = null; });
    }
    if (UNLOCKED) buildTabs();
  }, 250);

  function buildTabs() {
    const optCats = document.getElementById('optCats'), optRows = document.getElementById('optRows');
    if (!optCats || !optRows) return;
    if (!document.getElementById('r52_mods_btn')) {
      const b1 = document.createElement('button'); b1.id = 'r52_mods_btn'; b1.className = 'mi'; b1.textContent = '⚡ TUNER52'; b1.style.color = '#ef4444';
      const b2 = document.createElement('button'); b2.id = 'r52_fixes_btn'; b2.className = 'mi'; b2.textContent = '🛠️ FIXES'; b2.style.color = '#4ade80';
      optCats.appendChild(b1); optCats.appendChild(b2);
      b1.onclick = () => { tab = 'mods'; buildTabs(); }; b2.onclick = () => { tab = 'fixes'; buildTabs(); };
    }
    if (!document.getElementById('r52_mods_ui')) {
      const c1 = document.createElement('div'); c1.id = 'r52_mods_ui'; c1.style.cssText = 'display:none; flex-direction:column; gap:10px; width:100%;';
      const c2 = document.createElement('div'); c2.id = 'r52_fixes_ui'; c2.style.cssText = 'display:none; flex-direction:column; gap:10px; width:100%;';
      optRows.parentNode.appendChild(c1); optRows.parentNode.appendChild(c2);
    }
    const mUI = document.getElementById('r52_mods_ui'), fUI = document.getElementById('r52_fixes_ui');
    const mBtn = document.getElementById('r52_mods_btn'), fBtn = document.getElementById('r52_fixes_btn');
    if (tab === 'mods') { optRows.style.display = 'none'; fUI.style.display = 'none'; mUI.style.display = 'flex'; mBtn.classList.add('sel'); fBtn.classList.remove('sel'); renderMods(mUI); } 
    else if (tab === 'fixes') { optRows.style.display = 'none'; mUI.style.display = 'none'; fUI.style.display = 'flex'; fBtn.classList.add('sel'); mBtn.classList.remove('sel'); renderFixes(fUI); } 
    else { optRows.style.display = 'flex'; mUI.style.display = 'none'; fUI.style.display = 'none'; mBtn.classList.remove('sel'); fBtn.classList.remove('sel'); }
  }

  function btn(txt, col, act) { 
      const b = document.createElement('b'); b.textContent = txt; 
      b.style.cssText = `background:${col}; color:#fff; cursor:pointer; padding:6px 10px; border-radius:6px; font-size:11px; min-width:60px; text-align:center; flex:1; pointer-events:auto;`; 
      b.onclick = (e) => {
          if (b.dataset.busy === '1') return; // BUG 3 FIX
          act(e);
      }; 
      return b; 
  }

})();
