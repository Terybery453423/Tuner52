// ==UserScript==
// @name         Tuner52 (RALLY52) - Hybrid UI v3.0.2
// @namespace    https://github.com/Terybery453423
// @version      3.0.2
// @description  Tuner52 - Stealth mod menu for RALLY52.
// @match        https://rally52.com/*
// @grant        GM_xmlhttpRequest
// @grant        unsafeWindow
// @connect      raw.githubusercontent.com
// @connect      cdn.jsdelivr.net
// @connect      *
// @run-at       document-idle
// @license      GPL-3.0
// ==/UserScript==

(function(){
  'use strict';
  if (window.__r52hub) return; window.__r52hub = 1;

  const HUB_VERSION = '3.0.2'; 
  
  const REPO = 'https://raw.githubusercontent.com/Terybery453423/Tuner52/main/mods/';
  const ROOT_URL = 'https://raw.githubusercontent.com/Terybery453423/Tuner52/main/mod_hub.txt';
  const ALLOWED = [REPO, ROOT_URL];
  
  let SECRET = localStorage.getItem('r52_secret') === 'true';
  let UNLOCKED = localStorage.getItem('r52_unlocked') === 'true';
  let installed = JSON.parse(localStorage.getItem('r52_installed') || '[]');
  let active = JSON.parse(localStorage.getItem('r52_active') || '{}');
  let mods = [];
  let tab = 'mods';
  
  let updateAvailable = false;
  let latestVersion = HUB_VERSION;
  let latestScriptCode = '';

  const xhr = unsafeWindow.GM_xmlhttpRequest || GM_xmlhttpRequest;
  function fetchText(url) {
    return new Promise((res, rej) => {
      if (!ALLOWED.some(p => url.startsWith(p))) return rej('URL blocked');
      xhr({ method: 'GET', url: url + '?t=' + Date.now(), timeout: 10000,
        onload: r => r.status >= 200 && r.status < 300 ? res(r.responseText) : rej('HTTP ' + r.status),
        onerror: () => rej('Net error')
      });
    });
  }

  async function checkVersion() {
    try {
      const code = await fetchText(ROOT_URL);
      latestScriptCode = code;
      const match = code.match(/@version\s+([0-9.]+)/);
      if (match && match[1]) {
        latestVersion = match[1];
        if (latestVersion !== HUB_VERSION) {
          updateAvailable = true;
          const mUI = document.getElementById('r52_mods_ui') || document.getElementById('r52_modal_content');
          if (mUI) renderMods(mUI);
        }
      }
    } catch(e) {}
  }

  function inject(code) {
    return new Promise((resolve) => {
      const evt = 'r52_inject_' + Math.random().toString(36);
      const s = document.createElement('script');
      s.textContent = code + `\n;document.dispatchEvent(new CustomEvent('${evt}'));`;
      document.addEventListener(evt, () => resolve(), { once: true });
      (document.head || document.documentElement).appendChild(s);
      s.remove();
    });
  }

  inject(`
    window.__r52DownloadFile = function(url, name, onSuccess, onError) {
      const id = Math.random().toString(36);
      window._r52_dl_resolvers = window._r52_dl_resolvers || {};
      window._r52_dl_resolvers[id] = { onSuccess, onError };
      window.dispatchEvent(new CustomEvent('r52_hub_download', { detail: { url, name, id } }));
    };
    window.__r52ModUIToggles = window.__r52ModUIToggles || {};
    window.__r52ApplyUIToggle = function(id, visible) {
      if (window.__r52ModUIToggles[id]) window.__r52ModUIToggles[id](visible);
    };
  `);

  const page = unsafeWindow;
  const toPage = (t, d = '') => {
    const init = { detail: JSON.stringify({ t, d }) };
    page.dispatchEvent(new page.CustomEvent('r52-to-page', typeof cloneInto === 'function' ? cloneInto(init, page) : init));
  };
  let hubWs = null;
  page.addEventListener('r52-to-hub', (e) => {
    let m;
    try { m = JSON.parse(String(e.detail)); } catch (_) { return; }
    if (m.t === 'ping') toPage('pong');
    else if (m.t === 'connect') {
      if (hubWs) hubWs.close();
      hubWs = new WebSocket(String(m.d));
      hubWs.onopen = () => toPage('open');
      hubWs.onmessage = (msg) => toPage('message', msg.data);
      hubWs.onclose = () => toPage('close');
      hubWs.onerror = () => toPage('error', 'ws error');
    }
    else if (m.t === 'send') { if (hubWs && hubWs.readyState === 1) hubWs.send(String(m.d)); }
    else if (m.t === 'close') { if (hubWs) hubWs.close(); }
  });
  page.__tuner52Hub = 1;
  toPage('pong');

  window.addEventListener('r52_hub_download', (e) => {
    const { url, name, id } = e.detail;
    xhr({ method: 'GET', url: url, responseType: 'blob', timeout: 30000,
      onload: r => {
        if (r.status >= 200 && r.status < 300) {
          const objectUrl = URL.createObjectURL(r.response);
          if (unsafeWindow._r52_dl_resolvers[id]) { unsafeWindow._r52_dl_resolvers[id].onSuccess(objectUrl); delete unsafeWindow._r52_dl_resolvers[id]; }
        } else {
          if (unsafeWindow._r52_dl_resolvers[id]) { unsafeWindow._r52_dl_resolvers[id].onError('HTTP ' + r.status); delete unsafeWindow._r52_dl_resolvers[id]; }
        }
      },
      onerror: () => { if (unsafeWindow._r52_dl_resolvers[id]) { unsafeWindow._r52_dl_resolvers[id].onError('Net error'); delete unsafeWindow._r52_dl_resolvers[id]; } }
    });
  });

  async function loadMod(m) {
    if (active[m.id] === false) return;
    try {
      const code = await fetchText(m.url);
      await inject(code);
      const isHidden = localStorage.getItem('r52_ui_' + m.id) === 'true';
      if (isHidden) await inject(`if(window.__r52ApplyUIToggle) window.__r52ApplyUIToggle('${m.id}', false);`);
    } catch(e) { alert('❌ Failed to load: ' + m.name + '\nError: ' + e.message); }
  }

  checkVersion();
  
  (async () => {
    installed = [...new Set(installed)];
    if (!installed.length) return;
    try {
      const store = JSON.parse(await fetchText(REPO + 'store.json'));
      mods = mods.concat(store);
      store.forEach(m => { if (installed.includes(m.id) && active[m.id] !== false) loadMod(m); });
      if (SECRET) {
        const sec = JSON.parse(await fetchText(REPO + '45yllar.json'));
        mods = mods.concat(sec);
        sec.forEach(m => { if (installed.includes(m.id) && active[m.id] !== false) loadMod(m); });
      }
    } catch(e) {}
  })();

  // --- UI INJECTION LOGIC ---
  let taps = 0, tapT;

  function hijackSocials() {
    const tSocial = document.querySelector('.tSocial');
    if (!tSocial) return;
    const MY_TIKTOK = 'https://www.tiktok.com/@YOUR_TIKTOK';
    const MY_DISCORD = 'https://discord.gg/YOUR_INVITE';
    const ig = tSocial.querySelector('a[href*="instagram.com"]');
    const fb = tSocial.querySelector('a[href*="facebook.com"]');
    if (ig) ig.remove();
    if (fb) fb.remove();
    const tt = tSocial.querySelector('a[href*="tiktok.com"]');
    if (tt) { tt.href = MY_TIKTOK; tt.title = "TikTok"; }
    const dc = tSocial.querySelector('a[href*="discord.gg"]') || tSocial.querySelector('a[href*="discord.com"]');
    if (dc) { dc.href = MY_DISCORD; dc.title = "Discord"; }
  }

  function initFloatingUI() {
    if (document.getElementById('r52_floating_btn')) return;
    const btn = document.createElement('button');
    btn.id = 'r52_floating_btn';
    btn.textContent = '⚡';
    btn.style.cssText = 'position:fixed; bottom:20px; left:20px; z-index:2147483647; width:50px; height:50px; border-radius:50%; background:#ef4444; color:#fff; font-size:24px; border:2px solid #fff; cursor:pointer; box-shadow:0 4px 6px rgba(0,0,0,0.5); display:flex; align-items:center; justify-content:center;';
    document.body.appendChild(btn);

    const modal = document.createElement('div');
    modal.id = 'r52_modal';
    modal.style.cssText = 'position:fixed; top:0; left:0; width:100vw; height:100vh; background:rgba(0,0,0,0.8); z-index:2147483647; display:none; align-items:center; justify-content:center; backdrop-filter:blur(4px);';
    const inner = document.createElement('div');
    inner.style.cssText = 'background:#1a1a1a; width:90%; max-width:400px; height:80vh; border-radius:12px; border:1px solid #333; display:flex; flex-direction:column; overflow:hidden; font-family:monospace; color:#fff;';
    const header = document.createElement('div');
    header.style.cssText = 'padding:15px; background:#111; border-bottom:1px solid #333; display:flex; justify-content:space-between; align-items:center;';
    header.innerHTML = '<span style="font-weight:bold; font-size:16px; color:#ef4444;">TUNER52</span>';
    const closeBtn = document.createElement('button');
    closeBtn.textContent = '✕';
    closeBtn.style.cssText = 'background:none; border:none; color:#aaa; font-size:20px; cursor:pointer;';
    closeBtn.onclick = () => modal.style.display = 'none';
    header.appendChild(closeBtn);
    const content = document.createElement('div');
    content.id = 'r52_modal_content';
    content.style.cssText = 'flex:1; overflow-y:auto; padding:15px; display:flex; flex-direction:column; gap:10px;';
    inner.appendChild(header); inner.appendChild(content); modal.appendChild(inner); document.body.appendChild(modal);
    btn.onclick = () => { modal.style.display = 'flex'; renderMods(content); };
  }

  function updateNativeUI(optBox) {
    const mUI = document.getElementById('r52_mods_ui');
    const fUI = document.getElementById('r52_fixes_ui');
    const mBtn = document.getElementById('r52_mods_btn');
    const fBtn = document.getElementById('r52_fixes_btn');
    if (!mUI || !fUI || !mBtn || !fBtn) return;

    // Find the default game content container inside .optbox to hide it
    const gameRows = optBox.querySelectorAll('.orow, .oBody, .oContent');
    
    if (tab === 'mods') {
      gameRows.forEach(r => r.style.display = 'none');
      fUI.style.display = 'none'; mUI.style.display = 'flex';
      mBtn.classList.add('on'); fBtn.classList.remove('on');
      renderMods(mUI);
    } else if (tab === 'fixes') {
      gameRows.forEach(r => r.style.display = 'none');
      mUI.style.display = 'none'; fUI.style.display = 'flex';
      fBtn.classList.add('on'); mBtn.classList.remove('on');
      renderFixes(fUI);
    } else {
      gameRows.forEach(r => r.style.display = ''); // Restore default game UI
      mUI.style.display = 'none'; fUI.style.display = 'none';
      mBtn.classList.remove('on'); fBtn.classList.remove('on');
    }
  }

  function buildNativeTabs(optBox, optTabs) {
    if (!document.getElementById('r52_mods_btn')) {
      const b1 = document.createElement('button'); b1.id = 'r52_mods_btn'; b1.className = 'mi'; b1.setAttribute('role','tab'); b1.textContent = '⚡ TUNER52'; b1.style.color = '#ef4444';
      const b2 = document.createElement('button'); b2.id = 'r52_fixes_btn'; b2.className = 'mi'; b2.setAttribute('role','tab'); b2.textContent = '🛠️ FIXES'; b2.style.color = '#4ade80';
      
      optTabs.appendChild(b1); optTabs.appendChild(b2);
      
      b1.onclick = () => { tab = 'mods'; updateNativeUI(optBox); };
      b2.onclick = () => { tab = 'fixes'; updateNativeUI(optBox); };
    }
    if (!document.getElementById('r52_mods_ui')) {
      const c1 = document.createElement('div'); c1.id = 'r52_mods_ui'; c1.style.cssText = 'display:none; flex-direction:column; gap:10px; width:100%; padding: 10px; box-sizing: border-box;';
      const c2 = document.createElement('div'); c2.id = 'r52_fixes_ui'; c2.style.cssText = 'display:none; flex-direction:column; gap:10px; width:100%; padding: 10px; box-sizing: border-box;';
      optBox.appendChild(c1); optBox.appendChild(c2);
    }
    updateNativeUI(optBox);
  }

  setInterval(() => {
    hijackSocials(); 

    const optTitle = document.getElementById('oTitle');
    const optTabs = document.getElementById('oTabs');
    const optBox = document.querySelector('.optbox');
    
    // If the game's native settings menu is open
    if (optTitle && optTabs && optBox) {
      const fb = document.getElementById('r52_floating_btn');
      if (fb) fb.style.display = 'none';

      if (!optTitle.dataset.hooked) {
        optTitle.dataset.hooked = '1';
        // Tap the "SETTINGS" title 5 times to unlock
        optTitle.onclick = () => {
          taps++; clearTimeout(tapT); tapT = setTimeout(() => taps = 0, 1500);
          if (taps >= 5) { 
            taps = 0; 
            UNLOCKED = true; 
            localStorage.setItem('r52_unlocked', 'true'); 
            buildNativeTabs(optBox, optTabs); 
          }
        };
      }
      
      // If user clicks a default tab (not ours), hide our UI and reset
      if (!optTabs.dataset.hooked) {
        optTabs.dataset.hooked = '1';
        optTabs.addEventListener('click', e => {
          const btn = e.target.closest('button');
          if (btn && btn.id !== 'r52_mods_btn' && btn.id !== 'r52_fixes_btn') {
            tab = null;
            const mUI = document.getElementById('r52_mods_ui');
            const fUI = document.getElementById('r52_fixes_ui');
            if (mUI) mUI.style.display = 'none';
            if (fUI) fUI.style.display = 'none';
            // Restore default game UI rows
            optBox.querySelectorAll('.orow, .oBody, .oContent').forEach(r => r.style.display = '');
          }
        });
      }

      if (UNLOCKED) buildNativeTabs(optBox, optTabs);
    } else {
      initFloatingUI();
    }
  }, 250);

  function btn(txt, col, act) {
    const b = document.createElement('b'); b.textContent = txt;
    b.style.cssText = `background:${col}; color:#fff; cursor:pointer; padding:8px 12px; border-radius:6px; font-size:12px; min-width:70px; text-align:center; flex:1;`;
    b.onclick = act; return b;
  }

  function renderFixes(c) {
    c.innerHTML = '';
    const addFix = (title, desc, bTxt, col, act) => {
      const r = document.createElement('div'); r.className = 'orow'; r.style.padding = '10px'; r.style.background = '#222'; r.style.borderRadius = '8px'; r.style.display = 'flex'; r.style.justifyContent = 'space-between'; r.style.alignItems = 'center'; r.style.marginBottom = '8px';
      r.innerHTML = `<div style="display:flex; flex-direction:column; align-items:flex-start; gap:2px;"><span style="font-size:13px; color:#fff; font-weight:bold;">${title}</span><span style="font-size:10px; color:#aaa; text-transform:none;">${desc}</span></div>`;
      r.appendChild(btn(bTxt, col, act)); c.appendChild(r);
    };
    addFix('Reset Car Model', 'Restores default car.', 'RESET', '#3b82f6', () => {
      inject(`if(window.__r52ModCleanup){ if(window.__r52ModCleanup['glb_gltf_loader'])window.__r52ModCleanup['glb_gltf_loader'](); if(window.__r52ModCleanup['bridge'])window.__r52ModCleanup['bridge'](); } if(window.__r52unswap)window.__r52unswap(); alert('Car reset');`);
    });
    addFix('Clear Game Data', 'Wipes saves/ghosts.', 'CLEAR', '#ef4444', () => {
      if(!confirm('Wipe local saves?')) return;
      let cl=0, pats=[/^rd_202/,/^rd_ghost$/,/^rd_skids$/,/^rd_paint2_/,/^rd_art2_/,/^rd_wheels$/,/^rd_recent_cols$/,/^rd_rqdrop$/,/^rallydaily_/];
      for(let i=localStorage.length-1;i>=0;i--){ let k=localStorage.key(i); for(let p of pats){ if(p.test(k)){ localStorage.removeItem(k); cl++; break; } } }
      alert('Cleared '+cl+' keys.');
    });
  }

  async function renderMods(c) {
    if (c.dataset.loaded === '1') return;
    c.innerHTML = '<div style="color:#aaa; text-align:center;">Loading...</div>';
    let all = [];
    try { all = JSON.parse(await fetchText(REPO + 'store.json')); } catch(e) { c.innerHTML = '<div style="color:red;text-align:center;">Failed to load store</div>'; return; }
    if (SECRET) { try { const s = JSON.parse(await fetchText(REPO + '45yllar.json')); s.forEach(m=>m.isSecret=true); all = all.concat(s); } catch(e) {} }
    mods = all.filter((v,i,a)=>a.findIndex(t=>(t.id === v.id))===i);
    c.innerHTML = ''; c.dataset.loaded = '1';

    if (updateAvailable) {
      const banner = document.createElement('div');
      banner.style.cssText = 'background:#ef4444; color:#fff; padding:10px; border-radius:8px; font-size:12px; text-align:center; margin-bottom:5px; cursor:pointer;';
      banner.innerHTML = `⚠️ <b>UPDATE AVAILABLE (v${latestVersion})</b><br>Click to copy script.`;
      banner.onclick = async () => {
        await navigator.clipboard.writeText(latestScriptCode);
        banner.style.background = '#4ade80';
        banner.textContent = '✅ COPIED!';
      };
      c.appendChild(banner);
    }

    mods.forEach(m => {
      const isInstalled = installed.includes(m.id);
      const r = document.createElement('div'); 
      r.style.cssText = 'background:#222; border:1px solid #333; padding:12px; border-radius:8px; display:flex; flex-direction:column; gap:8px; margin-bottom:8px;';
      r.innerHTML = `<div style="font-size:14px; font-weight:bold; color:${m.isSecret?'#ef4444':'#fff'};">${m.name}</div><div style="font-size:11px; color:#aaa;">${m.desc}</div>`;

      if (m.options && m.options.length > 0) {
          const select = document.createElement('select');
          select.style.cssText = 'background:#111; color:#fff; border:1px solid #444; padding:6px; border-radius:4px; font-size:12px; width:100%;';
          const savedChoice = localStorage.getItem('r52_opt_' + m.id) || m.options[0];
          m.options.forEach(opt => {
              const o = document.createElement('option');
              o.value = opt; o.textContent = opt;
              if (opt === savedChoice) o.selected = true;
              select.appendChild(o);
          });
          select.onchange = async (e) => {
              localStorage.setItem('r52_opt_' + m.id, e.target.value);
              if (installed.includes(m.id) && active[m.id] !== false) {
                  inject(`if(window.__r52ModCleanup&&window.__r52ModCleanup['${m.id}'])window.__r52ModCleanup['${m.id}']();`);
                  await loadMod(m);
              }
          };
          r.appendChild(select);
      }

      const b = btn(isInstalled?'REMOVE':'INSTALL', isInstalled?'#ef4444':'#3b82f6', async (e) => {
        e.target.disabled = true;
        if (installed.includes(m.id)) {
          if (e.target.dataset.c === '1') {
            clearTimeout(e.target._t);
            installed = installed.filter(x => x !== m.id);
            delete active[m.id];
            localStorage.removeItem('r52_ui_' + m.id);
            localStorage.setItem('r52_installed', JSON.stringify(installed));
            localStorage.setItem('r52_active', JSON.stringify(active));
            inject(`if(window.__r52ModCleanup&&window.__r52ModCleanup['${m.id}'])window.__r52ModCleanup['${m.id}']();`);
            renderMods(c);
          } else {
            e.target.dataset.c = '1'; e.target.textContent = 'SURE?'; e.target.style.background = '#ef4444'; e.target.disabled = false;
            e.target._t = setTimeout(() => { e.target.dataset.c = '0'; e.target.textContent = 'REMOVE'; e.target.style.background = '#ef4444'; }, 3000);
          }
        } else {
          e.target.textContent = 'LOADING...'; e.target.style.background = '#5a2a2a';
          if (!installed.includes(m.id)) installed.push(m.id);
          active[m.id] = true;
          await loadMod(m);
          localStorage.setItem('r52_installed', JSON.stringify(installed));
          localStorage.setItem('r52_active', JSON.stringify(active));
          renderMods(c);
        }
      });
      r.appendChild(b); c.appendChild(r);
    });
  }
})();
